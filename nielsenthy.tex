\subsection{Parametrized traces and the Nielsen fixed point theorem}
For an allowable dynamical system $f\curvearrowright \tset\xrightarrow p\base$, we defined the associated Lefschetz index and Reidemeister trace
\begin{equation}
  \label{eq:nielsenthy-intro-characters}
  \lambda^f\colon (pi)_*(pi)^*\SS_\base\to \SS_\base\,
  \qquad\text{and}\qquad\rho_{f}\colon \SS_\base \to \SS [\cL^{f}\ani{\tset}[\base]]\,,
\end{equation}
where $i\colon\tset^f\to\tset$ denotes the inclusion of the fixed points.
The goal of this subsection is to establish the following relationship between $\lambda^f$ and $\rho_{f}$:
\begin{thm}\label{thm: lefschetz-nielsen}
  Let $f\curvearrowright \tset\xrightarrow p\base$ be an allowable dynamical system. There is a commutative diagram in the category of sheaves of spectra on $\base$:
  \[\begin{tikzcd}
      &  (pi)_*(pi)^!\SS_\base\\
      \SS_\base & {\SS [\cL^{f}\ani{\tset}[\base]]},
      \arrow["\mathfrak a", from=1-2, to=2-2]
      \arrow["{\lambda_f}", from=2-1, to=1-2]
      \arrow["{\rho_f}"', from=2-1, to=2-2]
    \end{tikzcd}\]
  where $\lambda_f$ denotes the $\SS_\base$-linear dual of $\lambda^f$. Furthermore,
  \begin{enumerate}[label=(\roman*)]
  \item $\mathfrak a$ is natural in proper maps of allowable dynamical systems; and
  \item for $f = \id$, the map $\mathfrak a$ is equivalent to the composition
    \[
      p_*p^!\SS_\base\xrightarrow\sim\SS\lbrack \ani{\tset}[\base]\rbrack\to \SS\lbrack \cL \ani{\tset}[\base]\rbrack,
    \]
    where the first map is the canonical equivalence and the second map is induced by the map $\ani{\tset}[\base]\to \cL \ani{\tset}[\base]$ adjoint to the projection $S_\base^1\times \ani{\tset}[\base]\to \ani{\tset}[\base]$.
  \end{enumerate}
\end{thm}
As an immediate consequence of this result, we will deduce the classical Nielsen fixed point theorem, see~\cref{subsec:classical-nielsen}.

The proof of~\cref{thm: lefschetz-nielsen} follows the same pattern as our treatment of the Lefschetz trace formula in the preceding section. Since $\rho_f$ can be defined as the trace of the morphism
\[
  \begin{tikzcd}
    \Sp_\base\arrow[d,equal] \arrow[r] & \Sp_\base^{\ani{\tset}[\base]}\arrow[d,"f^*"]\\
    \Sp_\base\arrow[r]\arrow[ru,Rightarrow, shorten <=5pt,shorten >=5pt] & \Sp_\base^{\ani{\tset}[\base]}
  \end{tikzcd}
\]
in the category of traceable endomorphisms $\Endtrl (\pPrlf\base )$, our goal is to factor this morphism through a traceable endomorphism that ``remembers the topology'' of the dynamical system. Such a factorization is provided by the assembly functor and Wall object of Bartels--Nikolaus (in the unparametrized case) and Volpe--Wolff--Ramzi (in the general case), as we explain in~\cref{subsec:factoring-reidemeister}.
\subsubsection{Factoring the Reidemeister trace}\label{subsec:factoring-reidemeister}
\begin{thm}
  Let $f\curvearrowright \tset\xrightarrow p\base$ be an allowable dynamical system. The lax commutative square
  \[
    \begin{tikzcd}
      \Sp_\base\arrow[d,equal] \arrow[r] & Sp_\base^{\ani{\tset}[\base]}\arrow[d,"f^*"]\\
      \Sp_\base\arrow[r]\arrow[ru,Rightarrow, shorten <=5pt,shorten >=5pt] & \Sp_\base^{\ani{\tset}[\base]}
    \end{tikzcd}
  \]
  admits a factorization
  \begin{equation}
    \label{eq:categorical-factorization-of-param-reidemeister}
    \begin{tikzcd}
      \Sp_\base & {\coShat_\base(\tset)} & \Sp_\base^{\ani{\tset}[\base]} \\
      \Sp_\base & {\coShat_\base(\tset)} & \Sp_\base^{\ani{\tset}[\base]}
      \arrow["{w_{\tset}}", from=1-1, to=1-2]
      \arrow["\id"', from=1-1, to=2-1]
      \arrow["{a_{\delta}}", from=1-2, to=1-3]
      \arrow["{f^?}", from=1-2, to=2-2]
      \arrow["f^{\ast}", from=1-3, to=2-3]
      \arrow[Rightarrow, from=2-2, to=1-3]
      \arrow["{w_{\tset}}"', from=2-1, to=2-2]
      \arrow[Rightarrow, from=2-1, to=1-2]
      \arrow["{a_{\delta}}"', from=2-2, to=2-3]
    \end{tikzcd}
  \end{equation}
  in $\Endtrl (\pPrlf{B})$, natural in proper maps of allowable dynamical systems.
\end{thm}
\begin{prop}\label{prop: lax calculation of trace on coshv hat}
  Let $f\curvearrowright \tset\xrightarrow p\base$ be an allowable dynamical system. 
  The map
  \[
    \TrcL (a_\delta)\colon \TrcL (f^?\curvearrowright\coShat_\base (\tset ))\too \TrcL(f^*\curvearrowright \Sp_\base^{\ani{\tset}[\base]})\simeq \SS [\cL^{f}\ani{\tset}[\base]]
  \]
  admits a factorization
  \begin{equation}
    \label{eq:lax-trace-on-cosheaves}
    \begin{tikzcd}
      &  (pi)_*(pi)^!\SS_\base\\
      \TrcL (f^?\curvearrowright\coShat_\base (\tset )) & {\SS [\cL^{f}\ani{\tset}[\base]]},
      \arrow[from=1-2, to=2-2]
      \arrow[from=2-1, to=1-2]
      \arrow[from=2-1, to=2-2]
    \end{tikzcd}
  \end{equation}
  in $\Sp_\base$, natural in proper maps of allowable dynamical systems. Furthermore, the composite
  \[
    \SS_\base\simeq\TrcL (\id\curvearrowright\Sp_\base)\to \TrcL (f^?\curvearrowright\coShat_\base (\tset ))\to (pi)_*(pi)^!\SS_\base
  \]
  is homotopic to $\lambda_f$.
\end{prop}
In order to prove~\cref{prop: lax calculation of trace on coshv hat}, we want to interpret the lax commutative square
  \[
    \begin{tikzcd}
      \coShat_\base(\tset )\arrow[d,"f^?"] \arrow[r,"a_\delta"] & Sp_\base^{\ani{\tset}[\base]}\arrow[d,"f^*"]\\
      \coShat_\base(\tset )\arrow[r,"a_\delta"]\arrow[ru,Rightarrow, shorten <=5pt,shorten >=5pt] & \Sp_\base^{\ani{\tset}[\base]}
    \end{tikzcd}
  \]
  in terms of an assembly map in a convenient closed symmetric monoidal category. As inspiration, recall that for a commutative ring $k$ there is an (often useful) equivalence between on one side the category of pairs $(V,t\colon V\to V)$, where $V$ is a $k$-module and $t$ is a $k$-linear map, and on the other side the category of modules over the polynomial ring $k\lbrack T\rbrack$. This equivalence is a consequence of the fact that $k\lbrack T\rbrack$ is the free associative (and coincidentally also commutative) $k$-algebra on a single generator, so a module structure $k\lbrack T\rbrack\to\End_k(V)$ is uniquely determined by the value of $T$. Similarly, we will consider the free associate $\Sp_\base$-algebra on a single dualizable object:
  \begin{thm}
    The forgetful functor
    \[
      \Fun(B\NN ,\pPrll\base)\too\pPrl\base
    \]
    has a left adjoint. Furthermore, this adjunction is monadic and symmetric monoidal, where $\Fun (B\NN ,\pPrll\base)$ is equipped with the pointwise Lurie tensor product.
  \end{thm}
\subsubsection{The Nielsen fixed point theorem}\label{subsec:classical-nielsen}
Let $\tset$ be a topological set that is compact Hausdorff and locally of constant shape. For a self-map $f\colon \tset\to \tset$, the so-called Nielsen fixed point theorem of Reidemeister~\cite{Reidemeister1936} and Wecken~\cite{Wecken1941a,Wecken1941b,Wecken1942} gives a lower bound on the number of fixed points of $f$ that only depends on the homotopy class of $f$. The lower bound is expressed in terms of the following numerical invariant:
\begin{defn}\label{defn: Nielsen number}
  The \tdef{Nielsen number} of $f$, denoted $\mdef{N_f}$, is the number of nonzero coordinates of the Reidemeister character $\rho_f\in\pi_0\SS\lbrack\cL^f\ani{\tset}\rbrack$ with respect to the standard basis
  \begin{equation}
    \label{eq:standard-basis-of-reidemeister-target}
    \pi_0\SS\lbrack\cL^f\ani{\tset}\rbrack\simeq\ZZ\lbrack\pi_0\cL^f\ani{\tset}\rbrack.
  \end{equation}
\end{defn}
By construction, $N_f$ depends only on the homotopy class of $f$.\footnote{In other approaches to Nielsen fixed point theory, the Nielsen number---or rather the Reidemeister trace---is often defined in terms of $\tset^f$, and one then has to do some work to show that it is homotopy-invariant. For us, the homotopy-invariance of the Nielsen number is congenital, and the work that we do instead goes towards showing that it relates to $\tset^f$, which is the content of~\cref{thm: lefschetz-nielsen}.}
\begin{prop}[Nielsen fixed point theorem]
  $\#\tset^f \geq N_f\,$.
\end{prop}
Since the Lefschetz trace is the sum of the coordinates of the Reidemeister trace, we have
\begin{equation}
  \label{eq:lefschetz-vs-nielsen-fixed-pt}
  \text{Lefschetz trace }\geq 1\Rightarrow N_f\geq 1.
\end{equation}
We can therefore view the Nielsen fixed point theorem as a refinement of the Lefschetz fixed point theorem. Note however that aside from~\eqref{eq:lefschetz-vs-nielsen-fixed-pt}, there is no general formula relating the Nielsen number and the Lefschetz trace. ()
\begin{proof}
  We may assume that $\# \tset^f < \infty$, as the statement is trivial otherwise. Consider the commutative diagram
  \[
    \begin{tikzcd}
      \tset^f\arrow[d,"\id"]\arrow[r,"i"] & \tset\arrow[d,"f"] \\
      \tset^f\arrow[r,"i"] & \tset,
    \end{tikzcd}
  \]
  where $i$ denotes the inclusion. By~\cref{thm: lefschetz-nielsen}, we get a commutative diagram
  \[
    \begin{tikzcd}
      \SS\lbrack \tset^f\rbrack \arrow[d] \arrow[r,"\sim"] & (pi)_*(pi)^!\SS\arrow[d,"\mathfrak a"] \\
      \SS\lbrack\cL \tset^f\rbrack\arrow[r] & \SS\lbrack\cL^f\tset\rbrack,
    \end{tikzcd}
  \]
  where the left vertical map is induced by the canonical map $\tset^f\to\cL \tset^f$, which is an equivalence since $\tset^f$ is discrete. On $\pi_0$, we thus find that the image of $\mathfrak a\colon (pi)_*(pi)^!\SS\to \SS\lbrack\cL^f\tset\rbrack$ is contained in the span of the basis vectors given by path components of $(x,c_x)\in\cL^f\tset$, where $x\in \tset^f$ and $c_x$ is the constant loop at $x$. It follows that $N_f\leq\# \tset^f$ as desired.
\end{proof}

\subsection{Cosheaves and assembly}

\begin{defn}[Bartels--Nikolaus]\label{defn: abstract assembly}
  Let $(\mathcal C,\otimes ,\1)$ be a closed symmetric monoidal category with internal mapping object $\Hom_{\mathcal C}({-},{-})$. Given objects $A$ and $B$ in $\mathcal C$ together with a map 
  \[
    \delta\colon \1\to A\otimes B,
  \]
  the associated {\em $\delta$-assembly map}, denoted $a_\delta$, is defined by the following string diagram 
  \begin{equation*}
    \label{eq:abstract-assembly}
    a_\delta =
    \begin{tikzpicture}[baseline=(current bounding box.center)]
      % Nodes
      \node (Bvee) at (-0.3,3.3) {$B^\vee$};
      \node (deltabox) at (1,2.5) [rectangle, draw, minimum width=0.6cm, minimum height=0.6cm] {$\delta$};
      \node (evbox) at (0,1) [rectangle, draw, minimum width=0.6cm, minimum height=0.6cm] {$\mathrm{ev}_\base$};
      \node (A) at (1.2,0) {$A$,};
      \node (B) at (0.3,1.8) {$B$};

      % Edges
      \draw (Bvee) -- (evbox.north -| -0.3,1.3);
      \draw (deltabox.south -| 0.8,2.5) to [out=270,in=90] (evbox.north -| 0.3,1.3);
      \draw (deltabox.south -| 1.2,2.5) -- (A);
    \end{tikzpicture}
  \end{equation*}
  where $B^\vee = \Hom_{\mathcal C}(B,1)$ and $\mathrm{ev}_\base\colon B^\vee\otimes B\to 1$ is the evaluation map, i.e. the counit of the adjunction between ${-}\otimes B$ and $\Hom_{\mathcal C}(B,{-})$.
\end{defn}
Let $\Prdual$ denote the category of dualizable stable categories with strongly continuous functors between them. The Lurie tensor product on presentable categories restricts to a symmetric monoidal structure on $\Prdual$. By \cite{Ramzi2024}, this symmetric monoidal structure is closed. We denote the resulting internal mapping object by $\Homd ({-},{-})$.
\begin{defn}[Bartels--Nikolaus]
  Let $\tset$ be a locally compact Hausdorff space. The category of \tdef{completed cosheaves} on $\tset$, denoted $\coShat (\tset )$, is given by
  \[
    \coShat (\tset ) = \Homd (\D{\tset},\Sp ).
  \]
\end{defn}
Although general objects of $\coShat (\tset )$ are quite mysterious, the definition does give us a description of the compact objects via
\[
  \coShat (\tset )^\omega\simeq\Map_{\Prdual}(\Sp ,\coShat (\tset ))
  \simeq\Map_{\Prdual}(\Sp,\Homd (\D{\tset},\Sp ))\simeq\Map_{\Prdual}(\D{\tset},\Sp),
\]
where the final equivalence used the adjunction between internal mapping object and $\otimes$. That is, the groupoid of compact objects $\coShat (\tset )^\omega$ is equivalent to the groupoid strongly continuous functors $\D{\tset}\to\Sp$, which is a subgroupoid of the more familiar groupoid of all colimit-preserving functors $\D{\tset}\to\Sp$. Of particular interest is the strongly continuous functor
\[
\tset_\sharp\colon\D{\tset}\to\Sp
\]
that is left adjoint to $\tset^*$, defined if $\tset$ is locally of singular shape. The corresponding compact object of $\coShat (\tset )$ is denoted $w_\tset$ and referred to as the \tdef{local Wall object} of $\tset$.

Another aspect of $\coShat (\tset )$ which is well-understood by hard work of Bartels--Nikolaus and Efimov is its value under localizing invariants:
\begin{thm}[Bartels--Efimov--Nikolaus]\label{thm: K theory of completed cosheaves}
  $K(\coShat (\tset ))\simeq \tset_*\tset^!K (\SS )$.
\end{thm}
Finally, we will need to introduce some notation for the functoriality of $\coShat (\tset )$. If $h\colon\tset\to\tset '$ is a proper map of locally compact Hausdorff spaces, we get as usual that the pullback $h^*\colon\D{\tset '}\to \D{\tset}$ is strongly continuous. Hence it induces a map on completed cosheaves that we denote
\[
h_?\colon \coShat (\tset )\to \coShat (\tset ').
\]
That is, the category of completed cosheaves is {\em covariantly functorial in proper maps}. Since $h_?$ is strongly continuous by construction, it admits a colimit-preserving right3 adjoint that we denote
\[
h^?\colon \coShat (\tset ')\to \coShat (\tset ).
\]
The reader may observe that there is an additional contravariant functoriality in etale maps, for instance, that one would might want to denote $h^{\text{?`}}$.


\medskip

Assume that $\tset$ is locally of singular shape and compact. Under these assumptions, we have a sequence of colimit-preserving functors
\[
  \D{\tset}\otimes\Sp^{\fundgroupoid (\tset )}\xrightarrow{\id\otimes\psi^*}
  \D{\tset}\otimes \D{\tset}\xrightarrow{\Delta^*}\D{\tset}\xrightarrow{\tset^*}
  \Sp, 
\]
where $\psi^*\colon \Sp^{\fundgroupoid (\tset )}\to\D{\tset}$ denotes the inclusion of local systems into all sheaves. Bartels and Nikolaus show that the composite $\D{\tset}\otimes\Sp^{\fundgroupoid (\tset )}\to\Sp$ admits a left adjoint
\[
\delta\colon \Sp\to \D{\tset}\otimes\Sp^{\fundgroupoid (\tset )},
\]
which is then a strongly continuous functor and thus a morphism in the category $\Prdual$. Specializing~\cref{defn: abstract assembly} to this situation, we thus get a strongly continuous functor
\begin{equation}
  \label{eq:BN-assembly}
  a_\delta\colon\coShat (\tset )\to \Sp^{\fundgroupoid (\tset )},
\end{equation}
the \tdef{Bartels--Nikolaus assembly functor}.
\begin{thm}[Bartels--Nikolaus]\label{thm: Bartels--Nikolaus geometric topology}
  Let $\tset$ be a compact Hausdorff space that is locally of constant shape.
  Then the functor~\eqref{eq:BN-assembly}
  \begin{enumerate}[label=(\roman*)]
  \item induces the Waldhausen assembly map on $K$-theory and
  \item maps the local Wall object $w_\tset$ to the constant sphere spectrum $\SS_{\fundgroupoid \tset}$.
  \end{enumerate}
\end{thm}
Part (ii) means that we can factor $(\fundgroupoid\tset )^*\colon\Sp\to\Sp^{\fundgroupoid\tset}$ through the Bartels--Nikolaus assembly functor. Better, we have
